Flutter — panel arrastrable genérico

class _DraggablePanel extends StatefulWidget {
  final String panelId;         // 'top' | 'mouse' | 'windows'
  final Offset initialOffset;   // posición base por sesión
  final Widget child;
  ...
}

class _DraggablePanelState extends State<_DraggablePanel> {
  Offset _off = Offset.zero;

  Widget build(BuildContext context) {
    return Positioned(
      left: widget.initialOffset.dx + _off.dx,
      top:  widget.initialOffset.dy + _off.dy,
      child: GestureDetector(
        // SOLO la manija arrastra — el resto del panel
        // conserva sus taps (botones, teclas) intactos.
        onPanUpdate: (d) => setState(() {
          _off = _clamp(_off + d.delta, screenSize, panelSize);
        }),
        child: Column(children: [
          _DragHandle(),   // ⠿ icono, cursor grab
          widget.child,    // contenido 1:1 del panel actual
        ]),
      ),
    );
  }
}

Kotlin — lista REAL de ventanas X11 (worker + xdotool)

// DesktopController: canal nuevo "listWindows"
fun listWindows(): List<X11Window> {
    val session = synchronized(lock) { activeSession } ?: return emptyList()
    return session.listWindows()  // spawn xdotool con wmEnv real
}

// DesktopSessionManager (allowlist binaria: xdotool, sin comandos libres)
fun listWindows(): List<X11Window> {
    val out = spawnCaptured(  // worker MSG_SPAWN → archivo tmp
        binary = File(usrDir, "bin/xdotool"),
        argv   = listOf("search", "--onlyvisible", "--name", ".*",
                        "getwindowname", "getwindowgeometry", "getwindowpid"),
        envp   = lastWmEnv,   // MISMO env del WM: DISPLAY, LD_LIBRARY_PATH
    )
    return parseXdotool(out)  // id + título + geometría + pid
}
